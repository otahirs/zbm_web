        function createNewRowForDay(otherTr){

            const template = otherTr.dataset.template;
            Object.assign(tr.dataset, otherTr.dataset);
            const formNamePrefix = `template[${template}][${otherTr.dataset.day}][${new Date().getTime()}]`;
            tr.innerHTML = `

            tr.querySelector(".delete").addEventListener("click", deleteRow);
  
            const dayTrList = document.querySelectorAll(`[data-template="${template}"][data-day="${otherTr.dataset.day}"]`);
            let appendAfterTr = dayTrList.item(dayTrList.length - 1);

            appendAfterTr.parentNode.insertBefore(tr, appendAfterTr.nextSibling);
            $(tr).find("select").multiselect();
            return tr;
        }

        function appendRowToDay(clickEvent) {
            if (document.querySelector(".multiselect-list.active") !== null) return;
            createNewRowForDay(clickEvent.target.parentNode);
            clickEvent.target.rowSpan = clickEvent.target.rowSpan + 1;
        }

        document.querySelectorAll(".day").forEach((day) => {
            day.addEventListener("click", appendRowToDay)
        })

        // rebuilt first coll of table with rowspan
        function rebuiltDayNames() {
            document.querySelectorAll(".day").forEach(e => e.parentNode.removeChild(e));
            const CZdays = ["pondělí", "úterý", "středa", "čtvrtek", "pátek", "sobota", "neděle"];
            const days = ['monday','tuesday','wednesday','thursday','friday','saturday', 'sunday'];
            const rows = document.querySelectorAll("table.edit-plan tr");
            let currDay = 0;
            rows.forEach(function(tr, i) {
                if (tr.dataset.day != days[currDay]) {
                    return;
                }
                if (tr.style.display === "none") {
                    if(rows[i+1]?.dataset.day == days[currDay]) {
                        return;
                    } else {
                        tr = createNewRowForDay(tr);
                    }
                    
                }
                let template = tr.dataset.template;
                let day = days[currDay];
                let dayName = document.createElement("td");
                dayName.className = "day";
                dayName.innerHTML = CZdays[currDay];
                dayName.rowSpan = document.querySelectorAll(`[data-template="${template}"][data-day="${day}"]:not([style*="display: none"]`).length;
                dayName.addEventListener("click", appendRowToDay);
                tr.prepend(dayName);
                currDay += 1;
                if (currDay % 7 == 0) {
                    currDay = 0;
                }
            });

        }

        // cleanup table

            const rows = document.querySelectorAll("table.edit-plan tr");
            rows.forEach(row => {

        // calendar of modes
        const calendarRows = document.getElementById("plan-calendar__rows");
        const templateNames = Array.from(document.querySelectorAll("#choose-template option")).map(o => o.value);

        function toMonday(value) {
            if (!value) return value;
            const [y, m, d] = value.split("-").map(Number);
            const date = new Date(Date.UTC(y, m - 1, d));
            date.setUTCDate(date.getUTCDate() - (date.getUTCDay() + 6) % 7);
            return date.toISOString().slice(0, 10);
        }

        function onCalendarDateChange(e) {
            const monday = toMonday(e.target.value);
            if (monday && monday !== e.target.value) {
                e.target.value = monday;
                notyf.success("Režim se mění v pondělí, datum posunuto na " + monday.split("-").reverse().join("."));
            }
        }

        let calendarRowCounter = calendarRows.querySelectorAll("tr").length;
        function addCalendarRow(from, template) {
            const i = calendarRowCounter++;
            const tr = document.createElement("tr");
            tr.className = "plan-calendar__row";
            const td1 = document.createElement("td");
            const input = document.createElement("input");
            input.type = "date";
            input.name = `calendar[${i}][from]`;
            input.value = from || "";
            input.addEventListener("change", onCalendarDateChange);
            td1.appendChild(input);
            const td2 = document.createElement("td");
            const select = document.createElement("select");
            select.name = `calendar[${i}][template]`;
            templateNames.forEach(n => {
                const opt = document.createElement("option");
                opt.value = n;
                opt.textContent = n;
                opt.selected = (n === template);
                select.appendChild(opt);
            });
            td2.appendChild(select);
            const td3 = document.createElement("td");
            td3.className = "plan-calendar__delete";
            td3.innerHTML = '<i class="fa fa-trash-o" aria-hidden="true"></i>';
            td3.addEventListener("click", () => tr.remove());
            tr.append(td1, td2, td3);
            calendarRows.appendChild(tr);
            if (template && !templateNames.includes(template)) {
                notyf.error(`Šablona „${template}“ neexistuje, vyberte jinou (${from})`);
            }
            return tr;
        }

        function sortCalendar() {
            Array.from(calendarRows.querySelectorAll("tr"))
                .sort((a, b) => (a.querySelector("input").value || "9999").localeCompare(b.querySelector("input").value || "9999"))
                .forEach(tr => calendarRows.appendChild(tr));
        }

        calendarRows.querySelectorAll("input[type=date]").forEach(i => i.addEventListener("change", onCalendarDateChange));
        calendarRows.querySelectorAll(".plan-calendar__delete").forEach(td => td.addEventListener("click", () => td.parentNode.remove()));
        document.getElementById("plan-calendar__add").addEventListener("click", () => addCalendarRow("", templateNames[0]));

        document.getElementById("plan-calendar__suggest").addEventListener("click", () => {
            const data = new FormData();
            data.append("year", document.getElementById("plan-calendar__year").value);
            $.ajax({
                url: "/php/plan/suggestcalendar",
                type: "POST",
                data: data,
                dataType: "text",
                processData: false,
                contentType: false,
                success: function (response) {
                    const text = String(response);
                    const result = JSON.parse(text.slice(text.indexOf("{"), text.lastIndexOf("}") + 1));
                    const existing = new Set(Array.from(calendarRows.querySelectorAll("input[type=date]")).map(i => i.value));
                    let added = 0;
                    result.rows.forEach(r => {
                        if (existing.has(r.from)) return;
                        addCalendarRow(r.from, r.template).classList.add("plan-calendar--suggested");
                        added++;
                    });
                    sortCalendar();
                    notyf.success(`Přidáno řádků: ${added}. ${result.notes.join(" ")}<br>Zkontrolujte a uložte.`);
                },
                error: function (xhr) {
                    notyf.error(xhr.responseText || "Neočekávaná chyba");
                }
            });
        });
